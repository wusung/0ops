```go
// ObserveDeployRunTransition records state transition.
func (m *Metrics) ObserveDeployRunTransition(stateFrom, stateTo, teamBucket string) {
m.deployRunTransitions.WithLabelValues(stateFrom, stateTo, teamBucket).Inc()
}

// ObserveDeployRunLeadTime records lead time from queued to live.
func (m *Metrics) ObserveDeployRunLeadTime(outcome, teamBucket string, duration time.Duration) {
m.deployRunLeadTime.WithLabelValues(outcome, teamBucket).Observe(duration.Seconds())
}

// ObservePreviewCreated increments preview creation counter.
func (m *Metrics) ObservePreviewCreated(teamBucket string) {
m.previewCreated.WithLabelValues(teamBucket).Inc()
}

// ObservePreviewConsumed records preview consumption.
func (m *Metrics) ObservePreviewConsumed(outcome, teamBucket string) {
m.previewConsumed.WithLabelValues(outcome, teamBucket).Inc()
}

// ObservePreviewConsumeDuration records time from preview to confirm.
func (m *Metrics) ObservePreviewConsumeDuration(teamBucket string, duration time.Duration) {
m.previewConsumeDuration.WithLabelValues(teamBucket).Observe(duration.Seconds())
}

// SetCloudflareConnectorsReady sets current connector count.
func (m *Metrics) SetCloudflareConnectorsReady(region string, count int) {
m.cloudflareConnectorsReady.WithLabelValues(region).Set(float64(count))
}
```

- [ ] **Step 2: Write recording rules ConfigMap**

```bash
cat > /home/foxdie/Projects/0ops/0ops/.worktrees/m2-impl/deploy/gitops/observability/prometheus-recording-rules.yaml << 'EOF'
apiVersion: v1
kind: ConfigMap
metadata:
  name: prometheus-recording-rules
  namespace: prometheus  # Adjust per deployment
data:
  recording-rules.yaml: |
    groups:
      - name: zeroops.recording
        interval: 30s
        rules:
          # Deploy run success rate (# SLI 4)
          - record: cluster:zeroops_deploy_run_terminal:rate5m
            expr: |
              sum(rate(zeroops_deploy_run_state_transitions_total{state_to="live"}[5m])) by (team_bucket) 
          
          # Deploy run failure rate (# SLI 4)
          - record: cluster:zeroops_deploy_run_failed:rate5m
            expr: |
              sum(rate(zeroops_deploy_run_state_transitions_total{state_to=~"failed|rolled_back"}[5m])) by (team_bucket)
          
          # API latency p95 read (# SLI 2)
          - record: cluster:zeroops_http_request_duration:p95_5m
            expr: |
              histogram_quantile(0.95, sum(rate(zeroops_http_request_duration_seconds_bucket{route!~".*:preview$"}[5m])) by (le, team_bucket))
          
          # API latency p95 preview (# SLI 3)
          - record: cluster:zeroops_http_request_duration_preview:p95_5m
            expr: |
              histogram_quantile(0.95, sum(rate(zeroops_http_request_duration_seconds_bucket{route=~".*:preview$"}[5m])) by (le, team_bucket))
          
          # Preview consumption rate (# SLI 8)
          - record: cluster:zeroops_preview_consumption:rate7d
            expr: |
              sum(increase(zeroops_preview_consumed_total{outcome=~"success|idempotent_replay"}[7d])) by (team_bucket)
              / 
              sum(increase(zeroops_preview_created_total[7d])) by (team_bucket)
          
          # Preview to confirm latency p50 (# SLI 9)
          - record: cluster:zeroops_preview_consume_duration:p50_7d
            expr: |
              histogram_quantile(0.50, sum(rate(zeroops_preview_consume_duration_seconds_bucket[7d])) by (le, team_bucket))
          
          # HTTP request error rate (# SLI 1)
          - record: cluster:zeroops_http_errors:rate28d
            expr: |
              sum(rate(zeroops_http_requests_total{status=~"5.."}[28d])) by (team_bucket)
              /
              sum(rate(zeroops_http_requests_total[28d])) by (team_bucket)
EOF
```

## Task 4: Create Prometheus alert rules ConfigMap

**Files:**
- Create: `deploy/gitops/observability/prometheus-alert-rules.yaml`

- [ ] **Step 1: Write alert rules ConfigMap**

```bash
cat > /home/foxdie/Projects/0ops/0ops/.worktrees/m2-impl/deploy/gitops/observability/prometheus-alert-rules.yaml << 'EOF'
apiVersion: v1
kind: ConfigMap
metadata:
  name: prometheus-alert-rules
  namespace: prometheus
data:
  alert-rules.yaml: |
    groups:
      - name: zeroops.alerts
        interval: 30s
        rules:
          # API Availability (SLI 1: 99.9%)
          - alert: ApiAvailabilityBurnRate
            expr: |
              (cluster:zeroops_http_errors:rate28d > 0.001)
            for: 5m
            annotations:
              summary: "API availability SLO burn rate high ({{ $value | humanizePercentage }})"
              
          # Deploy run success rate (SLI 4: > 85%)
          - alert: DeploySuccessRateLow
            expr: |
              (1 - (cluster:zeroops_deploy_run_failed:rate5m / (cluster:zeroops_deploy_run_terminal:rate5m + cluster:zeroops_deploy_run_failed:rate5m))) < 0.85
            for: 10m
            annotations:
              summary: "Deploy success rate below 85% threshold"
              
          # Preview consumption rate (SLI 8: > 80%)
          - alert: PreviewConsumptionLow
            expr: |
              cluster:zeroops_preview_consumption:rate7d < 0.80
            for: 1h
            annotations:
              summary: "Preview consumption rate below 80% ({{ $value | humanizePercentage }})"
              
          # Cloudflare tunnel down (SLI 7: 99.95%)
          - alert: CloudflareTunnelDown
            expr: |
              zeroops_cloudflare_tunnel_connectors_ready{region=~".*"} == 0
            for: 2m
            annotations:
              summary: "Cloudflare tunnel connector down for region {{ $labels.region }}"
EOF
```
