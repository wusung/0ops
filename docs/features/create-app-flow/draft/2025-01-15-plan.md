# Create App Flow (M2.1) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Refactor create_app from inline handler to service-based architecture with saga pattern, state machine, and comprehensive error handling.

**Architecture:** Implement preview.Action interface with SideEffects/Precheck/Execute methods. Coordinate 5 reversible/irreversible side effects across database, K3s, GitOps, ops_token signing, and GitHub Actions. Deploy_run state machine manages workflow lifecycle from queued → live with compensating rollback on failure.

**Tech Stack:** Go (idiomatic), Go testing (table-driven), database transactions, state machines, saga pattern

```go
package createapp

import "fmt"

// Deploy_run states per spec § 7.2
const (
StateQueued        = "queued"
StatePreparing     = "preparing"
StateBuilding      = "building"
StatePushing       = "pushing"
StateRendering     = "rendering"
StateSyncing       = "syncing"
StateLive          = "live"
StateCompensating  = "compensating"
StateRolledBack    = "rolled_back"
StateFailed        = "failed"
)

type StateMachine struct {
currentState string
}

func NewStateMachine(state string) *StateMachine {
return &StateMachine{currentState: state}
}

// Transition validates and applies state transitions per spec
func (sm *StateMachine) Transition(toState string) error {
// Valid transitions:
// queued → preparing
// preparing → building
// building → pushing
// pushing → rendering
// rendering → syncing
// syncing → live
// Any state → compensating (on error)
// compensating → rolled_back (success)
// compensating → failed (failure)

validTransitions := map[string][]string{
StateQueued:       {StatePreparing},
StatePreparing:    {StateBuilding},
StateBuilding:     {StatePushing},
StatePushing:      {StateRendering},
StateRendering:    {StateSyncing},
StateSyncing:      {StateLive},
StateCompensating: {StateRolledBack, StateFailed},
}

if allowed, ok := validTransitions[sm.currentState]; ok {
for _, s := range allowed {
if s == toState {
sm.currentState = toState
return nil
}
}
}

return fmt.Errorf("invalid transition from %s to %s", sm.currentState, toState)
}

func (sm *StateMachine) State() string {
return sm.currentState
}
```

package createapp

import (
"context"
"fmt"
)

type sagaStep struct {
name      string
forward   func(ctx context.Context) error
backward  func(ctx context.Context) error
reversible bool
}

func (s *Service) Execute(ctx context.Context, args *AppCreateArgs) (interface{}, error) {
// Spec § 6.2: Execute saga with reversible + irreversible stages

appID := generateAppID()
steps := []sagaStep{
{
name:       "db-insert",
reversible: true,
forward: func(ctx context.Context) error {
return s.dbInsertApp(ctx, appID, args)
},
backward: func(ctx context.Context) error {
return s.dbDeleteApp(ctx, appID)
},
},
{
name:       "k3s-namespace",
reversible: true,
forward: func(ctx context.Context) error {
return s.k3sCreateNamespace(ctx, appID, args.Slug)
},
backward: func(ctx context.Context) error {
return s.k3sDeleteNamespace(ctx, appID, args.Slug)
},
},
{
name:       "gitops-render-push",
reversible: true,
forward: func(ctx context.Context) error {
return s.gitopsRenderPush(ctx, appID, args)
},
backward: func(ctx context.Context) error {
return s.gitopsRevert(ctx, appID, args)
},
},
{
name:       "ops-token-sign",
reversible: false,
forward: func(ctx context.Context) error {
return s.signOpsToken(ctx, appID)
},
},
{
name:       "gha-dispatch",
reversible: false,
forward: func(ctx context.Context) error {
return s.ghaDispatchWorkflow(ctx, appID, args)
},
},
}

completed := 0
for i, step := range steps {
if err := step.forward(ctx); err != nil {
// Compensate: run backward in reverse order (only reversible)
for j := completed - 1; j >= 0; j-- {
if steps[j].reversible {
_ = steps[j].backward(ctx) // Best effort
}
}
return nil, fmt.Errorf("saga failed at %s: %w", step.name, err)
}
completed++
}

return map[string]string{"app_id": appID}, nil
}
