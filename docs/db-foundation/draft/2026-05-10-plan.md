# DB Foundation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立 `internal/server/db/` 的 Postgres 基礎設施，接上 `pgx + sqlc`，並用真實 Postgres smoke test 驗證第一個最小 team/membership repository。

**Architecture:** 以 `pgxpool` 管理連線生命週期，`sqlc` 只生成符合 ADR-0001 邊界的 team/membership 查詢，`ResolveTeamBySlug` 則作為 repository 內受限例外以手寫 `pgx` 實作。測試直接打 migration 後的 Postgres，不用 mock，不引入 HTTP/CLI/MCP。

**Tech Stack:** Go 1.25、`jackc/pgx/v5`、`sqlc`、Postgres 17、`testing`

```go
// internal/server/db/config.go
package db

import (
	"os"
	"time"
)

type Config struct {
	URL             string
	MaxConns        int32
	MinConns        int32
	MaxConnLifetime time.Duration
	MaxConnIdleTime time.Duration
}

func ConfigFromEnv() Config {
	return Config{
		URL:             os.Getenv("DATABASE_URL"),
		MaxConns:        10,
		MinConns:        0,
		MaxConnLifetime: 30 * time.Minute,
		MaxConnIdleTime: 5 * time.Minute,
	}
}
```

```go
// internal/server/db/db.go
package db

import (
	"context"
	"errors"

	"github.com/jackc/pgx/v5/pgxpool"
)

func NewPool(ctx context.Context, cfg Config) (*pgxpool.Pool, error) {
	if cfg.URL == "" {
		return nil, errors.New("database url is required")
	}

	poolCfg, err := pgxpool.ParseConfig(cfg.URL)
	if err != nil {
		return nil, err
	}

	poolCfg.MaxConns = cfg.MaxConns
	poolCfg.MinConns = cfg.MinConns
	poolCfg.MaxConnLifetime = cfg.MaxConnLifetime
	poolCfg.MaxConnIdleTime = cfg.MaxConnIdleTime

	pool, err := pgxpool.NewWithConfig(ctx, poolCfg)
	if err != nil {
		return nil, err
	}

	return pool, nil
}

func MustPing(ctx context.Context, pool *pgxpool.Pool) error {
	return pool.Ping(ctx)
}
```

- [ ] **Step 3: Write minimal repository implementation**

```go
package db

import (
	"context"
	"fmt"

	"github.com/google/uuid"
	"github.com/jackc/pgx/v5/pgxpool"

	sqlcgen "github.com/winshare/zeroops/internal/server/db/sqlc"
)

type Repository struct {
	pool    *pgxpool.Pool
	queries *sqlcgen.Queries
}

func NewRepository(pool *pgxpool.Pool) *Repository {
	return &Repository{
		pool:    pool,
		queries: sqlcgen.New(pool),
	}
}

func (r *Repository) ResolveTeamBySlug(ctx context.Context, slug string) (Team, error) {
	row := r.pool.QueryRow(ctx, `
SELECT id, slug, name, plan, archived_at
FROM team
WHERE slug = $1
`, slug)

	var team Team
	var id uuid.UUID
	if err := row.Scan(&id, &team.Slug, &team.Name, &team.Plan, &team.ArchivedAt); err != nil {
		return Team{}, err
	}
	team.ID = id.String()
	return team, nil
}

func (r *Repository) ListUserTeams(ctx context.Context, userID string, limit int32, afterSlug *string) ([]TeamMembership, error) {
	id, err := uuid.Parse(userID)
	if err != nil {
		return nil, fmt.Errorf("parse user id: %w", err)
	}

	rows, err := r.queries.ListUserTeams(ctx, sqlcgen.ListUserTeamsParams{
		UserID: id,
		Column2: afterSlug,
		Limit:  limit,
	})
	if err != nil {
		return nil, err
	}

	items := make([]TeamMembership, 0, len(rows))
	for _, row := range rows {
		items = append(items, TeamMembership{
			Team: Team{
				ID:         row.ID.String(),
				Slug:       row.Slug,
				Name:       row.Name,
				Plan:       row.Plan,
				ArchivedAt: row.ArchivedAt,
			},
			UserID:    row.UserID.String(),
			Role:      row.Role,
			JoinedAt:  row.JoinedAt,
			InvitedAt: row.InvitedAt,
		})
	}

	return items, nil
}

func (r *Repository) CheckTeamMembership(ctx context.Context, teamID string, userID string) (bool, error) {
	parsedTeamID, err := uuid.Parse(teamID)
	if err != nil {
		return false, fmt.Errorf("parse team id: %w", err)
	}
	parsedUserID, err := uuid.Parse(userID)
	if err != nil {
		return false, fmt.Errorf("parse user id: %w", err)
	}

	return r.queries.CheckTeamMembership(ctx, sqlcgen.CheckTeamMembershipParams{
		TeamID: parsedTeamID,
		UserID: parsedUserID,
	})
}
```

Standardize how contributors add new migrations without rewriting history or forgetting downstream regeneration steps.

1. `make migrate`
2. If query shapes changed: `make sqlc`

4. If schema affects other packages: `go test ./...`

## Rules

- Never commit a migration with only `Up`
- Never change a historical migration to “fix” a new problem
- Always regenerate sqlc when schema or query signatures change
- Always add or update tests for changed query behavior
